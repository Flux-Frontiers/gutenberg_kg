gutenberg_kg.image_gen

Image generation — local MLX or remote mflux-serve.

Two code paths: generate() — local Flux2Klein via mflux (Apple Silicon only) generate_via_server() — HTTP call to a running mflux-serve instance generate_auto() — server if GUTENKG_IMAGE_ENDPOINT / server_url set, else local fallback

Environment variables

GUTENKG_IMAGE_MODEL HuggingFace repo or mflux model name (default: mlx-community/flux2-klein-4b-4bit) GUTENKG_IMAGE_STEPS Inference steps (default: 4) GUTENKG_IMAGE_SIZE Default output size WIDTHxHEIGHT (default: 1536x1024) GUTENKG_IMAGE_ENDPOINT Base URL of a running mflux-serve instance (default: empty — use local generation) Example: http://localhost:8090 (mflux-server default)

 38def _parse_size(size: str | None) -> tuple[int, int] | None:
 39    """Parse an explicit ``"WIDTHxHEIGHT"`` string into an ``(width, height)`` pair.
 40
 41    :param size: Size string such as ``"768x512"`` (case-insensitive ``x``), or None.
 42    :returns: ``(width, height)`` when *size* parses to two positive ints, else None.
 43    """
 44    if not size:
 45        return None
 46    try:
 47        w_str, h_str = size.lower().split("x", 1)
 48        width, height = int(w_str), int(h_str)
 49    except (ValueError, AttributeError):
 50        return None
 51    if width <= 0 or height <= 0:
 52        return None
 53    return width, height

 56_DEFAULT_VLM_BASE_URL = "http://localhost:8080/v1"
 57_DEFAULT_VLM_MODEL = "Qwen3-4B-Instruct-2507-MLX-8bit"
 58
 59_VLM_SYSTEM = (
 60    "You are an expert art director. Given a passage of historical text, write a single concise "
 61    "image generation prompt (one paragraph, no bullet points, no quotation marks) that vividly "
 62    "describes the scene for a text-to-image model. Focus on visual elements: setting, lighting, "
 63    "figures, mood, and artistic style. Do NOT include any text, labels, captions, or words in "
 64    "the scene description. Output ONLY the prompt, nothing else."
 65)

 68def vlm_rewrite(
 69    corpus_text: str,
 70    *,
 71    base_url: str | None = None,
 72    model: str | None = None,
 73) -> tuple[str, str | None]:
 74    """Rewrite corpus prose into a visual image-generation prompt via a local VLM.
 75
 76    :param corpus_text: Historical text to rewrite as a visual scene description.
 77    :param base_url: OpenAI-compatible endpoint URL. Falls back to GUTENKG_VLM_ENDPOINT
 78                     or http://localhost:8080/v1.
 79    :param model: Model ID. Falls back to GUTENKG_VLM_MODEL or the Qwen3-30B default.
 80    :returns: (prompt, error) — prompt is the rewritten text (or corpus_text on failure);
 81              error is None on success or a short message describing the failure.
 82    """
 83    import re
 84
 85    try:
 86        from openai import OpenAI
 87    except ImportError:
 88        return corpus_text, "openai package not installed"
 89
 90    url = base_url or os.environ.get("GUTENKG_VLM_ENDPOINT", _DEFAULT_VLM_BASE_URL)
 91    mdl = model or os.environ.get("GUTENKG_VLM_MODEL", _DEFAULT_VLM_MODEL)
 92
 93    try:
 94        client = OpenAI(base_url=url, api_key="not-needed")
 95        response = client.chat.completions.create(
 96            model=mdl,
 97            messages=[
 98                {"role": "system", "content": _VLM_SYSTEM},
 99                {"role": "user", "content": corpus_text},
100            ],
101            max_tokens=300,
102            temperature=0.7,
103            extra_body={"think": False, "chat_template_kwargs": {"enable_thinking": False}},
104        )
105        raw = (response.choices[0].message.content or "").strip()
106        # Strip any <think>…</think> blocks the model emitted despite being asked not to.
107        cleaned = re.sub(r"<think>.*?</think>", "", raw, flags=re.DOTALL).strip()
108        return (cleaned or corpus_text), None
109    except Exception as exc:  # noqa: BLE001
110        return corpus_text, str(exc)

113# Module-level model cache for MCP server reuse
114_cached_model = None
115_cached_model_name: str | None = None
116
117
118def _load_model(model_name: str):
119    """Load Flux2Klein, reusing the cached instance when model_name is unchanged."""
120    global _cached_model, _cached_model_name
121    if _cached_model is not None and _cached_model_name == model_name:
122        return _cached_model
123
124    from mflux.models.flux2.variants.txt2img.flux2_klein import Flux2Klein
125
126    _cached_model = Flux2Klein(model_path=model_name)
127    _cached_model_name = model_name
128    return _cached_model

131def generate(
132    prompt: str,
133    *,
134    size: str | None = None,
135    seed: int | None = None,
136    output_path: str | Path | None = None,
137    model_name: str | None = None,
138    steps: int | None = None,
139) -> PILImage:
140    """Generate an image locally via Flux2Klein (Apple Silicon / mflux).
141
142    :param prompt: Text description of the image to generate.
143    :param size: Output size ``"WIDTHxHEIGHT"`` (default: GUTENKG_IMAGE_SIZE or 1536x1024).
144    :param seed: Random seed for reproducibility (random if omitted).
145    :param output_path: If given, save the PNG here in addition to returning it.
146    :param model_name: Override the HF model repo (default: mlx-community/flux2-klein-4b-4bit).
147    :param steps: Override inference steps (default: 4).
148    :returns: PIL Image.
149    """
150    model_name = model_name or os.environ.get("GUTENKG_IMAGE_MODEL", _DEFAULT_MODEL)
151    steps = steps or int(os.environ.get("GUTENKG_IMAGE_STEPS", _DEFAULT_STEPS))
152    seed = seed if seed is not None else random.randint(0, 2**31 - 1)
153    size = size or os.environ.get("GUTENKG_IMAGE_SIZE", _DEFAULT_SIZE)
154    width, height = _parse_size(size) or _DEFAULT_DIMS
155
156    model = _load_model(model_name)
157    result = model.generate_image(
158        seed=seed,
159        prompt=prompt,
160        width=width,
161        height=height,
162        guidance=1.0,
163        num_inference_steps=steps,
164        scheduler="flow_match_euler_discrete",
165    )
166
167    pil_image: PILImage = result.image
168
169    if output_path is not None:
170        pil_image.save(str(output_path))
171
172    return pil_image

175def generate_via_server(
176    prompt: str,
177    *,
178    server_url: str,
179    size: str | None = None,
180    seed: int | None = None,
181    steps: int | None = None,
182) -> PILImage:
183    """Generate an image by calling a running mflux-serve HTTP server.
184
185    Requires only httpx + pillow — safe to call from Linux containers or any
186    environment without mflux installed.
187
188    :param prompt: Text description of the image to generate.
189    :param server_url: Base URL of the mflux-serve instance, e.g. http://localhost:8088.
190    :param size: Output size ``"WIDTHxHEIGHT"`` (default: GUTENKG_IMAGE_SIZE or 1536x1024).
191    :param seed: Optional integer seed for reproducibility.
192    :param steps: Override inference steps (default: GUTENKG_IMAGE_STEPS or 4).
193    :returns: PIL Image decoded from the server response.
194    """
195    import httpx
196    from PIL import Image
197
198    steps = steps or int(os.environ.get("GUTENKG_IMAGE_STEPS", _DEFAULT_STEPS))
199    size = size or os.environ.get("GUTENKG_IMAGE_SIZE", _DEFAULT_SIZE)
200    width, height = _parse_size(size) or _DEFAULT_DIMS
201
202    payload: dict = {
203        "prompt": prompt,
204        "n": 1,
205        "size": f"{width}x{height}",
206        "num_inference_steps": steps,
207        "response_format": "b64_json",
208    }
209    if seed is not None:
210        payload["seed"] = seed
211
212    resp = httpx.post(
213        server_url.rstrip("/") + "/v1/images/generations",
214        json=payload,
215        timeout=httpx.Timeout(connect=5.0, read=300.0, write=10.0, pool=5.0),
216    )
217    resp.raise_for_status()
218    b64 = resp.json()["data"][0]["b64_json"]
219    return Image.open(BytesIO(base64.b64decode(b64)))

222def generate_auto(
223    prompt: str,
224    *,
225    server_url: str | None = None,
226    size: str | None = None,
227    seed: int | None = None,
228    steps: int | None = None,
229    model_name: str | None = None,
230) -> PILImage:
231    """Generate an image, preferring a remote server and falling back to local mflux.
232
233    Resolution order:
234      1. *server_url* argument (non-empty string)
235      2. GUTENKG_IMAGE_ENDPOINT environment variable
236      3. Local generate() — requires mflux on Apple Silicon
237
238    :param prompt: Text description of the image to generate.
239    :param server_url: Override server URL; pass None to use env var or local.
240    :param size: Output size ``"WIDTHxHEIGHT"`` (default: GUTENKG_IMAGE_SIZE or 1536x1024).
241    :param seed: Optional integer seed for reproducibility.
242    :param steps: Override inference steps.
243    :param model_name: Local model override (ignored when using server).
244    :returns: PIL Image.
245    """
246    url = server_url or os.environ.get("GUTENKG_IMAGE_ENDPOINT", "")
247    if url:
248        return generate_via_server(prompt, server_url=url, size=size, seed=seed, steps=steps)
249    return generate(prompt, size=size, seed=seed, steps=steps, model_name=model_name)
