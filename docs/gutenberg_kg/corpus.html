corpus.py — Library API for GutenbergKG corpus-level operations.

Provides path-parametrised functions for building corpus status reports and snapshot dicts. All functions accept explicit path arguments so they can be called from any context — CLI, tests, or kgrag adapters — without depending on package-level path constants.

 58def _sqlite_counts(path: str | None) -> tuple[int, int]:
 59    """Return (nodes, edges) from a graph.sqlite, or (0, 0) on any error."""
 60    if not path:
 61        return 0, 0
 62    try:
 63        with sqlite3.connect(path) as con:
 64            nodes = con.execute("SELECT COUNT(*) FROM nodes").fetchone()[0]
 65            edges = con.execute("SELECT COUNT(*) FROM edges").fetchone()[0]
 66        return nodes, edges
 67    except Exception:  # noqa: BLE001
 68        return 0, 0

 71def _count_authors(corpus_root: Path) -> int:
 72    """Count author subdirectories in ``<corpus_root>/authors/``."""
 73    authors_dir = corpus_root / "authors"
 74    if not authors_dir.is_dir():
 75        return 0
 76    return sum(1 for p in authors_dir.iterdir() if p.is_dir())

 79def _git_info(repo_root: Path) -> dict[str, str]:
 80    """Return branch, short commit, and full commit from git."""
 81
 82    def _run(*args: str) -> str:
 83        """Run a git command in *repo_root* and return its stripped stdout, or "unknown" on failure.
 84
 85        :param args: Command and arguments (e.g. ``"git", "rev-parse", "HEAD"``).
 86        :return: Stripped stdout, or ``"unknown"`` if the command fails.
 87        """
 88        try:
 89            return subprocess.check_output(
 90                list(args), cwd=repo_root, text=True, stderr=subprocess.DEVNULL
 91            ).strip()
 92        except Exception:  # noqa: BLE001
 93            return "unknown"
 94
 95    return {
 96        "branch": _run("git", "rev-parse", "--abbrev-ref", "HEAD"),
 97        "commit": _run("git", "rev-parse", "--short", "HEAD"),
 98        "commit_full": _run("git", "rev-parse", "HEAD"),
 99    }

107def collect_genre_stats(registry_path: Path) -> list[dict[str, Any]]:
108    """Aggregate per-genre stats from the KGRAG registry.
109
110    Returns a list of dicts with keys: ``corpus``, ``label``, ``books``,
111    ``nodes``, ``edges``.  Only SQLite files that exist on disk are counted.
112
113    :param registry_path: Path to the KGRAG registry SQLite file.
114    :return: Per-genre stat dicts in ``GENRE_LABELS`` order.
115    :raises OSError: If the registry cannot be opened.
116    """
117    results: list[dict[str, Any]] = []
118    try:
119        reg = sqlite3.connect(str(registry_path))
120    except Exception as exc:  # noqa: BLE001
121        raise OSError(f"Cannot open registry {registry_path}: {exc}") from exc
122
123    kg_map: dict[str, str | None] = {}
124    for row in reg.execute("SELECT id, sqlite_path FROM kg_entries"):
125        kg_map[row[0]] = row[1]
126
127    for corpus_key, label in GENRE_LABELS.items():
128        row = reg.execute("SELECT kg_ids FROM corpora WHERE name = ?", (corpus_key,)).fetchone()
129        if row is None:
130            results.append(
131                {
132                    "corpus": corpus_key,
133                    "label": label,
134                    "books": 0,
135                    "nodes": 0,
136                    "edges": 0,
137                }
138            )
139            continue
140
141        kg_ids: list[str] = json.loads(row[0])
142        total_nodes = total_edges = live_books = 0
143        for kid in kg_ids:
144            sqlite_path = kg_map.get(kid)
145            if sqlite_path and Path(sqlite_path).exists():
146                n, e = _sqlite_counts(sqlite_path)
147                total_nodes += n
148                total_edges += e
149                live_books += 1
150
151        results.append(
152            {
153                "corpus": corpus_key,
154                "label": label,
155                "books": live_books,
156                "nodes": total_nodes,
157                "edges": total_edges,
158            }
159        )
160
161    reg.close()
162    return results

196        "platform": platform.platform(),

245def snapshot_save(
246    registry_path: Path,
247    repo_root: Path,
248    corpus_root: Path,
249    output: Path | None = None,
250) -> tuple[Path, dict[str, Any]]:
251    """Build and persist a corpus snapshot.
252
253    :param registry_path: KGRAG registry SQLite path.
254    :param repo_root: Root of the gutenberg_kg repository.
255    :param corpus_root: Corpus data directory.
256    :param output: Override output file path
257        (default: ``<corpus_root>/.snapshots/snapshot-<ts>.json``).
258    :return: Tuple of ``(saved_path, snapshot_dict)``.
259    """
260    snap = snapshot_build(registry_path, repo_root, corpus_root)
261    safe_ts = snap["timestamp"].replace(":", "-").replace("+", "").split(".")[0]
262
263    if output is not None:
264        out_path = output
265    else:
266        snapshots_dir = corpus_root / ".snapshots"
267        snapshots_dir.mkdir(parents=True, exist_ok=True)
268        out_path = snapshots_dir / f"snapshot-{safe_ts}.json"
269
270    out_path.parent.mkdir(parents=True, exist_ok=True)
271    out_path.write_text(json.dumps(snap, indent=2) + "\n", encoding="utf-8")
272    return out_path, snap

275def snapshot_list(snapshots_dir: Path) -> list[dict[str, Any]]:
276    """Load and return all snapshots, oldest first.
277
278    :param snapshots_dir: Directory containing ``snapshot-*.json`` files.
279    :return: List of snapshot dicts; empty if directory not found or empty.
280    """
281    if not snapshots_dir.is_dir():
282        return []
283    result = []
284    for p in sorted(snapshots_dir.glob("snapshot-*.json")):
285        try:

287        except Exception:  # noqa: BLE001
288            pass
289    return result

292def snapshot_show(snapshots_dir: Path, snapshot: str | None = None) -> dict[str, Any]:
293    """Return the full dict for a snapshot.
294
295    :param snapshots_dir: Directory containing ``snapshot-*.json`` files.
296    :param snapshot: Filename or timestamp prefix to match
297        (default: most recent).
298    :return: Snapshot dict, or ``{}`` if not found.
299    """
300    if not snapshots_dir.is_dir():
301        return {}
302    paths = sorted(snapshots_dir.glob("snapshot-*.json"))
303    if not paths:
304        return {}
305    if snapshot:
306        matches = [p for p in paths if snapshot in p.name]
307        if not matches:
308            return {}
309        path = matches[-1]
310    else:
311        path = paths[-1]
312    try:
313        return json.loads(path.read_text(encoding="utf-8"))
314    except Exception:  # noqa: BLE001
315        return {}

318def snapshot_diff(
319    snapshots_dir: Path,
320    a: str | None = None,
321    b: str | None = None,
322) -> dict[str, Any]:
323    """Return a structured diff between two snapshots.
324
325    :param snapshots_dir: Directory containing ``snapshot-*.json`` files.
326    :param a: First snapshot filename or prefix (default: second-to-last).
327    :param b: Second snapshot filename or prefix (default: most recent).
328    :return: Diff dict with ``a``, ``b``, ``totals``, ``changed_genres``; or
329        ``{"error": ...}`` if fewer than two snapshots exist.
330    """
331    if not snapshots_dir.is_dir():
332        return {"error": "No snapshots directory found."}
333    paths = sorted(snapshots_dir.glob("snapshot-*.json"))
334    if len(paths) < 2:
335        return {"error": "Need at least two snapshots to diff."}
336
337    def _resolve(name: str | None, fallback: Path) -> dict:
338        """Load the snapshot matching *name*, or *fallback* if *name* is None.
339
340        :param name: Snapshot filename or timestamp prefix to match; None uses *fallback*.
341        :param fallback: Path to load when *name* is None or matches nothing.
342        :return: Parsed snapshot dict, or ``{}`` if not found or unreadable.
343        """
344        if name is None:
345            try:
346                return json.loads(fallback.read_text(encoding="utf-8"))
347            except Exception:  # noqa: BLE001
348                return {}
349        matches = [p for p in paths if name in p.name]
350        if not matches:
351            return {}
352        try:
353            return json.loads(matches[-1].read_text(encoding="utf-8"))
354        except Exception:  # noqa: BLE001
355            return {}
356
357    snap_a = _resolve(a, paths[-2])
358    snap_b = _resolve(b, paths[-1])
359    ta, tb = snap_a.get("totals", {}), snap_b.get("totals", {})
360
361    def _delta(key: str) -> dict[str, int]:
362        """Build a before/after/delta dict for a totals *key*.
363
364        :param key: Key to compare between the two snapshots' ``totals`` dicts.
365        :return: Dict with ``before``, ``after``, and ``delta`` values.
366        """
367        va, vb = ta.get(key, 0), tb.get(key, 0)
368        return {"before": va, "after": vb, "delta": vb - va}
369
370    genre_a = {g["corpus"]: g for g in snap_a.get("genres", [])}
371    genre_b = {g["corpus"]: g for g in snap_b.get("genres", [])}
372    changed = []
373    for corpus_key in sorted(set(genre_a) | set(genre_b)):
374        ga = genre_a.get(corpus_key, {"books": 0, "nodes": 0, "edges": 0})
375        gb = genre_b.get(corpus_key, {"books": 0, "nodes": 0, "edges": 0})
376        if ga["books"] != gb["books"] or ga["nodes"] != gb["nodes"]:
377            changed.append(
378                {
379                    "corpus": corpus_key,
380                    "books": {"before": ga["books"], "after": gb["books"]},
381                    "nodes": {
382                        "before": ga["nodes"],
383                        "after": gb["nodes"],
384                        "delta": gb["nodes"] - ga["nodes"],
385                    },
386                    "edges": {
387                        "before": ga["edges"],
388                        "after": gb["edges"],
389                        "delta": gb["edges"] - ga["edges"],
390                    },
391                }
392            )
393
394    return {
395        "a": {
396            "timestamp": snap_a.get("timestamp"),
397            "version": snap_a.get("version"),
398            "commit": snap_a.get("commit"),
399        },
400        "b": {
401            "timestamp": snap_b.get("timestamp"),
402            "version": snap_b.get("version"),
403            "commit": snap_b.get("commit"),
404        },
405        "totals": {
406            "books": _delta("books"),
407            "nodes": _delta("nodes"),
408            "edges": _delta("edges"),
409        },
410        "changed_genres": changed,
411    }

420    """Corpus-aware snapshot manager backed by ``kg_utils.snapshots.SnapshotManager``.
421
422    Snapshots are keyed by git tree hash, stored as ``<key>.json`` alongside a
423    ``manifest.json`` index in ``snapshots_dir``.  Use :meth:`capture` to build
424    a snapshot from the live corpus, then :meth:`save_snapshot` to persist it.
425
426    :param snapshots_dir: Directory where snapshot JSON files are stored.
427    :param registry_path: KGRAG registry SQLite path.
428    :param repo_root: Repository root (for git metadata).
429    :param corpus_root: Corpus data root (contains ``authors/`` etc.).
430    """
431
432    def __init__(
433        self,
434        snapshots_dir: Path | str,
435        *,
436        registry_path: Path,
437        repo_root: Path,
438        corpus_root: Path,
439    ) -> None:
440        """Initialize the manager and store the paths used to capture snapshots.
441
442        :param snapshots_dir: Directory where snapshot JSON files are stored.
443        :param registry_path: KGRAG registry SQLite path.
444        :param repo_root: Repository root (for git metadata).
445        :param corpus_root: Corpus data root (contains ``authors/`` etc.).
446        """
447        super().__init__(snapshots_dir, package_name="gutenberg-kg")
448        self.registry_path = registry_path
449        self.repo_root = repo_root

452    def capture(
453        self,
454        version: str | None = None,
455        branch: str | None = None,
456        graph_stats_dict: dict[str, Any] | None = None,
457        tree_hash: str = "",
458        hotspots: list[dict[str, Any]] | None = None,
459        issues: list[str] | None = None,
460        **extra_metrics: Any,
461    ) -> Any:
462        """Build a corpus snapshot from the live registry.
463
464        :param version: Version string; auto-detected from package if None.
465        :param branch: Git branch name; auto-detected if None.
466        :param graph_stats_dict: Ignored; corpus metrics are computed from the registry.
467        :param tree_hash: Git tree hash; auto-detected if not provided.
468        :param hotspots: Unused; accepted for LSP compatibility.
469        :param issues: Unused; accepted for LSP compatibility.
470        :param extra_metrics: Passed through to the base ``capture`` call.

478            "total_authors": _count_authors(self.corpus_root),

480        }

483            branch=branch,
484            graph_stats_dict=metrics,

489    def save_snapshot(self, snapshot: Any, *, force: bool = False) -> Any:  # type: ignore[override]
490        """Persist snapshot, guarding against degenerate (zero-book) state.
491
492        :param snapshot: Snapshot to persist.
493        :param force: If True, always create a new history entry.
494        :raises ValueError: If the snapshot has 0 books.
495        """
496        m = snapshot.metrics if isinstance(snapshot.metrics, dict) else {}
497        if m.get("total_books", 0) == 0:
498            raise ValueError(
499                "Refusing to save snapshot with 0 books. "
500                "Ensure the corpus has ingested books before capturing a snapshot."
501            )
502        return super().save_snapshot(snapshot, force=force)

504    def _compute_delta_from_metrics(
505        self, new_m: dict[str, Any], old_m: dict[str, Any]
506    ) -> dict[str, Any]:
507        """Compute corpus delta including books and authors.
508
509        :param new_m: Newer metrics dict.
510        :param old_m: Older metrics dict.
511        :return: Delta dict with nodes, edges, books, authors.
512        """
513        return {
514            "nodes": new_m.get("total_nodes", 0) - old_m.get("total_nodes", 0),
515            "edges": new_m.get("total_edges", 0) - old_m.get("total_edges", 0),
516            "books": new_m.get("total_books", 0) - old_m.get("total_books", 0),
517            "authors": new_m.get("total_authors", 0) - old_m.get("total_authors", 0),
518        }

520    def _metrics_changed(self, new_metrics: dict[str, Any], old_metrics: dict[str, Any]) -> bool:
521        """Return True if any corpus metric changed.
522
523        :param new_metrics: Newer metrics dict.
524        :param old_metrics: Older metrics dict.
525        """
526        return (
527            new_metrics.get("total_nodes", 0) != old_metrics.get("total_nodes", 0)
528            or new_metrics.get("total_books", 0) != old_metrics.get("total_books", 0)
529            or new_metrics.get("total_authors", 0) != old_metrics.get("total_authors", 0)
530        )

319def snapshot_diff(
320    snapshots_dir: Path,
321    a: str | None = None,
322    b: str | None = None,
323) -> dict[str, Any]:
324    """Return a structured diff between two snapshots.
325
326    :param snapshots_dir: Directory containing ``snapshot-*.json`` files.
327    :param a: First snapshot filename or prefix (default: second-to-last).
328    :param b: Second snapshot filename or prefix (default: most recent).
329    :return: Diff dict with ``a``, ``b``, ``totals``, ``changed_genres``; or
330        ``{"error": ...}`` if fewer than two snapshots exist.
331    """
332    if not snapshots_dir.is_dir():
333        return {"error": "No snapshots directory found."}
334    paths = sorted(snapshots_dir.glob("snapshot-*.json"))
335    if len(paths) < 2:
336        return {"error": "Need at least two snapshots to diff."}
337
338    def _resolve(name: str | None, fallback: Path) -> dict:
339        """Load the snapshot matching *name*, or *fallback* if *name* is None.
340
341        :param name: Snapshot filename or timestamp prefix to match; None uses *fallback*.
342        :param fallback: Path to load when *name* is None or matches nothing.
343        :return: Parsed snapshot dict, or ``{}`` if not found or unreadable.
344        """
345        if name is None:
346            try:
347                return json.loads(fallback.read_text(encoding="utf-8"))
348            except Exception:  # noqa: BLE001
349                return {}
350        matches = [p for p in paths if name in p.name]
351        if not matches:
352            return {}
353        try:
354            return json.loads(matches[-1].read_text(encoding="utf-8"))
355        except Exception:  # noqa: BLE001
356            return {}
357
358    snap_a = _resolve(a, paths[-2])
359    snap_b = _resolve(b, paths[-1])
360    ta, tb = snap_a.get("totals", {}), snap_b.get("totals", {})
361
362    def _delta(key: str) -> dict[str, int]:
363        """Build a before/after/delta dict for a totals *key*.
364
365        :param key: Key to compare between the two snapshots' ``totals`` dicts.
366        :return: Dict with ``before``, ``after``, and ``delta`` values.
367        """
368        va, vb = ta.get(key, 0), tb.get(key, 0)
369        return {"before": va, "after": vb, "delta": vb - va}
370
371    genre_a = {g["corpus"]: g for g in snap_a.get("genres", [])}
372    genre_b = {g["corpus"]: g for g in snap_b.get("genres", [])}
373    changed = []
374    for corpus_key in sorted(set(genre_a) | set(genre_b)):
375        ga = genre_a.get(corpus_key, {"books": 0, "nodes": 0, "edges": 0})
376        gb = genre_b.get(corpus_key, {"books": 0, "nodes": 0, "edges": 0})
377        if ga["books"] != gb["books"] or ga["nodes"] != gb["nodes"]:
378            changed.append(
379                {
380                    "corpus": corpus_key,
381                    "books": {"before": ga["books"], "after": gb["books"]},
382                    "nodes": {
383                        "before": ga["nodes"],
384                        "after": gb["nodes"],
385                        "delta": gb["nodes"] - ga["nodes"],
386                    },
387                    "edges": {
388                        "before": ga["edges"],
389                        "after": gb["edges"],
390                        "delta": gb["edges"] - ga["edges"],
391                    },
392                }
393            )
394
395    return {
396        "a": {
397            "timestamp": snap_a.get("timestamp"),
398            "version": snap_a.get("version"),
399            "commit": snap_a.get("commit"),
400        },
401        "b": {
402            "timestamp": snap_b.get("timestamp"),
403            "version": snap_b.get("version"),
404            "commit": snap_b.get("commit"),
405        },
406        "totals": {
407            "books": _delta("books"),
408            "nodes": _delta("nodes"),
409            "edges": _delta("edges"),
410        },
411        "changed_genres": changed,
412    }
