gutenberg_kg.serve.Chat

Chat.py — GutenbergKG Chat Interface

Streamlit chat UI for the corpus-gutenberg KGRAG worker. Searches the consolidated DocKG (241 books, 20 genres) and 4 DiaryKG temporal indices, and optionally synthesises answers via a local Ollama / oMLX model.

Run standalone (worker must be running first): gutenkg chat

Or via docker compose: docker compose --profile chat up

 40# Synthesis models to hide from the dropdown. Reasoning models like Agents-A1
 41# emit their chain-of-thought as plain prose in the response body — not as
 42# strippable `<think>` tags, and unaffected by the `enable_thinking:false` flag —
 43# so on RAG prompts the answer truncates inside the thinking and the UI shows raw
 44# reasoning instead of an answer. Also excludes non-chat utilities (document
 45# converters, embedding models). Matched case-insensitively as substrings.
 46_MODEL_BLOCKLIST: tuple[str, ...] = (
 47    "agents-a1",  # reasoning agent — unstrippable "Thinking Process:" prose
 48    "deepseek-r1",  # R1 reasoning model
 49    "gpt-oss",  # reasoning model (harmony channels leak into content)
 50    "markitdown",  # document-to-markdown converter, not a chat model
 51    "embed",  # embedding models (nomic-embed, mxbai-embed, qwen3-embedding)
 52)

 55def _is_synth_model(model_id: str) -> bool:
 56    """Return ``True`` if a model id is usable for RAG synthesis (not blocklisted).
 57
 58    :param model_id: Model id reported by the backend.
 59    :returns: ``False`` for reasoning/non-chat models unsuited to concise RAG.
 60    """
 61    lid = model_id.lower()
 62    return not any(pat in lid for pat in _MODEL_BLOCKLIST)

165def _score_bar(score: float, width: int = 80) -> str:
166    """Render an inline HTML bar visualising a similarity score, colour-coded by magnitude.
167
168    :param score: Similarity score in ``[0, 1]``.
169    :param width: Bar width in pixels.
170    :returns: HTML markup for the bar plus a numeric label.
171    """
172    pct = min(int(score * 100), 100)
173    color = "#27AE60" if score >= 0.7 else "#F39C12" if score >= 0.4 else "#E74C3C"
174    return (
175        f"<div style='display:inline-block;vertical-align:middle;"
176        f"width:{width}px;height:8px;background:var(--secondary-background-color);border-radius:4px;overflow:hidden;'>"
177        f"<div style='width:{pct}%;height:100%;background:{color};'></div></div>"
178        f"&nbsp;<small style='color:var(--text-color);opacity:0.6;font-size:10px;'>{score:.3f}</small>"
179    )

182def _preview(text: str, n: int = 220) -> tuple[str, bool]:
183    """Truncate text to roughly ``n`` characters at a word boundary.
184
185    :param text: Text to truncate.
186    :param n: Maximum length before truncation.
187    :returns: Tuple of ``(preview_text, was_truncated)``.
188    """
189    text = (text or "").strip()
190    if len(text) <= n:
191        return text, False
192    cut = text[:n].rsplit(" ", 1)[0].rstrip(" ,;:—-")
193    return cut + "…", True

196def _render_hit_card(hit: dict) -> None:
197    """Render a single search hit as a styled HTML card with an expandable full passage.
198
199    :param hit: Hit dictionary as returned by the worker (kg/node metadata, score, content).
200    """
201    kg_kind = hit.get("kg_kind", "").lower().split(".")[-1]
202    kg_name = hit.get("kg_name", "")
203    node_kind = hit.get("kind", "")
204    score = float(hit.get("score", 0.0))
205    source = hit.get("source_path") or "—"
206    content = hit.get("content") or hit.get("summary") or ""
207    preview, truncated = _preview(content, 220)
208
209    author = hit.get("author") or ""
210    title = hit.get("title") or ""
211    genre = hit.get("genre") or ""
212    meta_parts = [x for x in [genre, author, title] if x]
213    meta_line = " · ".join(meta_parts) if meta_parts else hit.get("name", "")
214
215    border_color = _KG_KIND_COLOR.get(kg_kind, "#555")
216    esc_preview = html.escape(preview)
217    details = ""
218    if truncated:
219        esc_full = html.escape(content).replace("\n", "<br>")
220        details = (
221            "<details style='margin-top:6px;'>"
222            "<summary style='cursor:pointer;color:#4A90D9;font-size:12px;'>📖 Full passage</summary>"
223            f"<div style='color:var(--text-color);font-size:13px;margin-top:6px;"
224            f"line-height:1.55;'>{esc_full}</div>"
225            "</details>"
226        )
227
228    st.markdown(
229        f"""
230        <div style="background:var(--secondary-background-color);border-left:4px solid {
231            border_color
232        };
233                    border-radius:6px;padding:10px 14px;margin-bottom:6px;">
234          {_kg_kind_badge(kg_kind, kg_name)}
235          &nbsp;
236          {_node_kind_badge(node_kind)}
237          &nbsp;&nbsp;
238          <b style="font-size:14px;color:var(--text-color);">{html.escape(meta_line)}</b>
239          <br>
240          <span style="color:var(--text-color);opacity:0.55;font-size:11px;
241                       font-family:monospace;">📄 {html.escape(source)}</span>
242          &nbsp;&nbsp;
243          {_score_bar(score)}
244          {
245            "<br><span style='color:var(--text-color);opacity:0.8;font-size:12px;'>"
246            + esc_preview
247            + "</span>"
248            if esc_preview
249            else ""
250        }
251          {details}
252        </div>
253        """,
254        unsafe_allow_html=True,
255    )

292def _query_worker(
293    query: str,
294    *,
295    worker_url: str,
296    corpus: str,
297    k: int,
298    min_score: float,
299    semantic_floor: float,
300    synthesize: bool,
301    secret: str,
302    model: str = "",
303    backend: str = "",
304) -> dict:
305    """Route a corpus query through the worker and return the raw result payload.
306
307    :param query: Natural-language query string.
308    :param worker_url: Base URL of the KGRAG worker.
309    :param corpus: Corpus scope, e.g. ``"all"``, ``"diary"``, or a genre name.
310    :param k: Number of hits to request.
311    :param min_score: Drop hits below this similarity score.
312    :param semantic_floor: Discard a KG entirely if its best hit is below this.
313    :param synthesize: Whether to also request a synthesised narrative answer.
314    :param secret: Shared secret for the worker (if configured).
315    :param model: Override model ID for synthesis.
316    :param backend: Synthesis backend (``"omlx"``, ``"ollama"``, ``"openai"``).
317    :returns: The worker's raw JSON response as a dict.
318    """
319    return WorkerClient(worker_url, secret).query(
320        query,
321        corpus=corpus,
322        k=k,
323        min_score=min_score,
324        semantic_floor=semantic_floor,
325        synthesize=synthesize,
326        model=model,
327        backend=backend,
328    )

331@st.cache_data(ttl=300, show_spinner=False)
332def _fetch_stats(worker_url: str) -> dict:
333    """Fetch live corpus totals from the worker's ``stats`` op (cached 5 min).
334
335    :param worker_url: Base URL of the KGRAG worker.
336    :returns: The worker's stats dict, or ``{}`` if the worker is unreachable so
337              the header degrades gracefully before the worker is up.
338    """
339    try:
340        resp = httpx.post(
341            f"{worker_url.rstrip('/')}/runsync",
342            json={"input": {"op": "stats"}},
343            timeout=10.0,
344        )
345        resp.raise_for_status()
346        payload = resp.json()
347    except httpx.HTTPError:
348        return {}
349    return payload.get("output", payload)

352@st.cache_data(ttl=300, show_spinner=False)
353def _corpus_options(worker_url: str) -> list[str]:
354    """Build the corpus-scope dropdown from the worker's live genre list.
355
356    ``all`` (DocKG + diaries) and ``diary`` (diaries only) always bookend the
357    list; the middle is every DocKG genre reported by the worker's ``list_genres``
358    op. The ``diaries`` genre is folded into the ``diary`` scope, so it is
359    excluded here. Falls back to ``["all", "diary"]`` if the worker is offline.
360
361    :param worker_url: Base URL of the KGRAG worker.
362    :returns: Ordered corpus-scope options for the sidebar selectbox.
363    """
364    try:
365        resp = httpx.post(
366            f"{worker_url.rstrip('/')}/runsync",
367            json={"input": {"op": "list_genres"}},
368            timeout=10.0,
369        )
370        resp.raise_for_status()
371        payload = resp.json()
372    except httpx.HTTPError:
373        return ["all", "diary"]
374    genres = (payload.get("output", payload)).get("genres", [])
375    names = sorted(g["genre"] for g in genres if g.get("genre") and g["genre"] != "diaries")
376    return ["all", *names, "diary"]

379@st.cache_data(ttl=60, show_spinner=False)
380def _fetch_models(worker_url: str, secret: str, backend: str = "") -> tuple[list[str], str]:
381    """Fetch the available model list and default model from the worker (cached 60s).
382
383    :param worker_url: Base URL of the KGRAG worker.
384    :param secret: Shared secret for the worker (if configured).
385    :param backend: Synthesis backend to query models for.
386    :returns: Tuple of ``(model_ids, default_model_id)``. Reasoning and non-chat
387              models (see ``_MODEL_BLOCKLIST``) are filtered out.
388    """
389    models, default = WorkerClient(worker_url, secret).list_models(backend=backend)
390    models = [m for m in models if _is_synth_model(m)]
391    if default and not _is_synth_model(default):
392        default = models[0] if models else ""
393    return models, default

401def _result_to_markdown(result: dict) -> str:
402    """Render a query result dict as a downloadable Markdown document.
403
404    :param result: Worker query result (question, corpus, synthesis, hits).
405    :returns: Markdown text summarising the answer and source passages.
406    """
407    lines = ["# GutenbergKG — Result", ""]
408    if result.get("query"):
409        lines += [f"**Question:** {result['query']}", ""]
410    if result.get("corpus") and result["corpus"] != "all":
411        lines += [f"**Corpus:** {result['corpus']}", ""]
412    if result.get("synthesis"):
413        lines += ["## Answer", "", result["synthesis"], ""]
414        if result.get("model"):
415            lines += [f"_Model: {result['model']}_", ""]
416    hits = result.get("hits", [])
417    if hits:
418        lines += [f"## Source passages ({len(hits)})", ""]
419        for h in hits:
420            author = h.get("author") or ""
421            title = h.get("title") or h.get("name") or "passage"
422            src = h.get("source_path") or "—"
423            head = " · ".join(x for x in [title, author] if x) or src
424            lines += [f"### {head}  ·  score {h.get('score', 0):.3f}", ""]
425            lines += [(h.get("content") or h.get("summary") or "").strip(), ""]
426    return "\n".join(lines)

429def _build_image_prompt(result: dict) -> str:
430    """Distil a result into a concise image-generation prompt (≤800 chars)."""
431    if result.get("synthesis"):
432        return result["synthesis"][:800]
433    hits = result.get("hits", [])[:3]
434    parts = [h.get("content") or h.get("summary") or "" for h in hits]
435    return " ".join(p.strip() for p in parts if p.strip())[:800]

447def _render_assistant_turn(result: dict) -> None:
448    """Render an assistant chat turn: synthesis (or warning), stats caption, and hit cards.
449
450    :param result: Worker query result to render.
451    """
452    hits = result.get("hits", [])
453    synthesis = result.get("synthesis")
454    synthesis_error = result.get("synthesis_error")
455
456    if not hits:
457        st.warning("No passages matched — try different wording or lower the min score.")
458        return
459
460    if synthesis:
461        st.markdown(synthesis)
462        if result.get("model"):
463            st.caption(f"🤖 {result['model']}")
464    elif synthesis_error:
465        st.warning(
466            f"Answer generation failed — **{synthesis_error}**\n\n"
467            "Check that Ollama/oMLX is running and reachable."
468        )
469    else:
470        st.info("Answer generation off — see source passages below.")
471
472    _parts = [
473        f"📊 {result.get('total_hits', len(hits))} passages · {result.get('kgs_queried', 0)} KGs queried"
474    ]
475    if result.get("search_ms") is not None:
476        _parts.append(f"search {result['search_ms']:,} ms")
477    if result.get("synthesis_ms") is not None:
478        _parts.append(f"synthesis {result['synthesis_ms']:,} ms")
479    st.caption(" · ".join(_parts))
480
481    with st.expander(f"📄 Source passages ({len(hits)})", expanded=not bool(synthesis)):
482        for hit in hits:
483            _render_hit_card(hit)
