gutenberg_kg.serve.handler

KGRAG handler — GutenbergKG corpus.

Serves semantic search over the full Project Gutenberg corpus baked into this image at /workspace/gutenberg/: .dockg/ — consolidated DocKG (241 books, 20 genres, 1.3M nodes) diaries/*/diarykg/ — 4 DiaryKG temporal indices (Pepys, Evelyn, Boswell)

Implements the RunPod serverless API (--rp_serve_api) so it can be driven by the Chat.py Streamlit UI, curl, or any compatible HTTP client.

Volume layout (baked in at build time)

/workspace/gutenberg/ .dockg/ graph.sqlite lancedb/ catalog.json diaries/ /.diarykg/ graph.sqlite lancedb/

Environment variables

GUTENBERG_ROOT Root of the baked-in bundle. Default: /workspace/gutenberg EMBED_MODEL Sentence-transformer model ID. Default: BAAI/bge-small-en-v1.5 HANDLER_SECRET Optional shared secret. Requests must include {"secret": ""}. VLLM_ENDPOINT_URL Optional OpenAI-compatible endpoint for synthesis (oMLX/Ollama/vLLM). VLLM_API_KEY Bearer token for the synthesis endpoint. Omit for Ollama. VLLM_MODEL Model ID. Default: Qwen3-8B-MLX-4bit SYNTH_MAX_K Max snippets fed to synthesis. Default: 12

Request schema

{ "query": str — natural-language query (required) "secret": str — required when HANDLER_SECRET is set "corpus": str — "all" | "gutenberg" | "diary" | (default: "all") is any genre name, e.g. "philosophy", "sacred-texts" "k": int — top-k hits (default: 8) "min_score": float — drop hits below this score (default: 0.0) "semantic_floor": float — discard KG if best hit is below this (default: 0.0) "synthesize": bool — call vLLM for a generated answer (default: false) "model": str — override VLLM_MODEL for this request "op": str — "models" returns {"models": [...], "default": ...} "stats" returns {"books", "genres", "diaries", "nodes", "edges", "embed_model"} "list_genres" returns [{"genre", "book_count"}, ...] "list_books" (needs "genre") returns its books "get_chapters" (needs "genre", "book") returns its chapter list "get_chapter" (needs "genre", "book", "section_id") returns chapter text }

150def _bootstrap_registry():
151    """Register the consolidated DocKG and every DiaryKG in the KGRegistry at startup.
152
153    Populates the module-level ``_KG_SQLITE`` and ``_DIARY_TABLES`` caches as a
154    side effect, opening each diary's LanceDB table for semantic-first search.
155
156    :returns: The populated ``KGRegistry`` instance.
157    """
158    from kg_rag.corpus_registry import CorpusRegistry
159    from kg_rag.primitives import CorpusEntry, KGEntry, KGKind
160    from kg_rag.registry import KGRegistry
161
162    REGISTRY_PATH.parent.mkdir(parents=True, exist_ok=True)
163    reg = KGRegistry(db_path=REGISTRY_PATH)
164    corp_reg = CorpusRegistry(db_path=REGISTRY_PATH)
165
166    # --- DocKG (consolidated prose + verse) ---
167    if not _DOCKG_SQLITE.exists():
168        print(f"[bootstrap] WARNING: DocKG not found at {_DOCKG_SQLITE}")
169        print("[bootstrap]   Run 'gutenkg build-corpus' then rebuild the image.")
170    else:
171        entry = KGEntry(
172            id=str(uuid.uuid4()),
173            name="gutenberg",
174            kind=KGKind.GUTENBERG,
175            repo_path=GUTENBERG_ROOT,
176            venv_path=Path("/usr"),
177            sqlite_path=_DOCKG_SQLITE,
178            lancedb_path=_DOCKG_LANCEDB,
179            created_at=datetime.now(UTC),
180            updated_at=datetime.now(UTC),
181        )
182        reg.register(entry)
183        _KG_SQLITE["gutenberg"] = _DOCKG_SQLITE
184        print(f"[bootstrap] registered gutenberg dockg ({_DOCKG_SQLITE})")
185
186    # --- DiaryKG indices (one per diary) ---
187    corp_reg.create(CorpusEntry(name="diaries", description="Project Gutenberg — diary corpora"))
188    n_diaries = 0
189    if _DIARIES_ROOT.exists():
190        for diary_dir in sorted(_DIARIES_ROOT.iterdir()):
191            if not diary_dir.is_dir() or diary_dir.name.startswith("."):
192                continue
193            diarykg_dir = diary_dir / ".diarykg"
194            sqlite = diarykg_dir / "graph.sqlite"
195            lancedb = diarykg_dir / "lancedb"
196            if not sqlite.exists():
197                print(f"[bootstrap] skipping {diary_dir.name} — no .diarykg/graph.sqlite")
198                continue
199            slug = _diary_slug(diary_dir.name)
200            entry = KGEntry(
201                id=str(uuid.uuid4()),
202                name=slug,
203                kind=KGKind.DIARY,
204                repo_path=diary_dir,
205                venv_path=Path("/usr"),
206                sqlite_path=sqlite,
207                lancedb_path=lancedb,
208                metadata={"source_file": _source_file_for(diary_dir)},
209                created_at=datetime.now(UTC),
210                updated_at=datetime.now(UTC),
211            )
212            reg.register(entry)
213            corp_reg.add_kg("diaries", entry.id)
214            _KG_SQLITE[slug] = sqlite
215            # Open the diary's vector store (sqlite-vec preferred, LanceDB fallback).
216            try:
217                src = _open_vector_source(diarykg_dir, label=f"diary {slug}")
218                if src is not None:
219                    _DIARY_TABLES[slug] = src
220            except Exception as exc:  # noqa: BLE001
221                print(f"[bootstrap] WARNING: could not open diary store {slug}: {exc}")
222            n_diaries += 1
223            print(f"[bootstrap] registered diary: {slug}")
224    else:
225        print("[bootstrap] no diaries/ directory found")
226    print(f"[bootstrap] diaries corpus: {n_diaries} KG(s)")
227
228    return reg

231def _open_vector_source(dockg_dir, *, label: str):
232    """Open a vector source for a ``.dockg`` dir — sqlite-vec if present, else LanceDB.
233
234    Prefers the exact ``vectors.sqlite`` store (see the sqlite-vec migration);
235    falls back to the LanceDB ``dockg_nodes`` table for un-converted corpora, so
236    the worker keeps serving during the transition.
237
238    :param dockg_dir: The ``.dockg`` directory holding ``vectors.sqlite`` / ``lancedb``.
239    :param label: Human label for startup logging.
240    :returns: A ``SqliteVecBackend`` or an open LanceDB table, or ``None``.
241    """
242    vectors = dockg_dir / "vectors.sqlite"
243    lancedb_dir = dockg_dir / "lancedb"
244    if vectors.exists():
245        from kg_utils.vector_backend import SqliteVecBackend
246
247        be = SqliteVecBackend(vectors, dim=384, meta_columns=_VEC_META, check_same_thread=False)
248        be.open(wipe=False)
249        print(f"[startup] {label}: sqlite-vec store ({be.count()} vectors)")
250        return be
251    if lancedb_dir.exists():
252        import lancedb
253
254        db = lancedb.connect(str(lancedb_dir))
255        names = list(db.table_names())
256        table = "dockg_nodes" if "dockg_nodes" in names else (names[0] if names else None)
257        if table is None:
258            print(f"[startup] WARNING: no lancedb table in {lancedb_dir}")
259            return None
260        tbl = db.open_table(table)
261        print(f"[startup] {label}: LanceDB table {table} ({tbl.count_rows()} rows)")
262        return tbl
263    print(f"[startup] WARNING: no vector store for {label} in {dockg_dir}")
264    return None

273def _open_dockg_store() -> None:
274    """Open the consolidated DocKG SQLite for FTS5/BM25 lexical search.
275
276    The lexical channel only activates when the corpus carries a ``nodes_fts``
277    index (built by ``dockg reindex-fts`` / a recent ``dockg build``).  When it
278    is absent the store stays ``None`` and retrieval falls back to pure dense
279    cosine ranking, so older corpora keep working unchanged.
280    """
281    global _DOCKG_STORE
282    if not _DOCKG_SQLITE.exists():
283        return
284    try:
285        from doc_kg.store import GraphStore
286
287        store = GraphStore(_DOCKG_SQLITE)
288        if not store.has_fts():
289            print(
290                f"[startup] WARNING: no FTS5 index in {_DOCKG_SQLITE}; lexical "
291                "channel disabled (run 'dockg reindex-fts' or rebuild the corpus)"
292            )
293            store.close()
294            return
295        _DOCKG_STORE = store
296        print("[startup] opened DocKG lexical (FTS5) index — hybrid retrieval enabled")
297    except Exception as exc:  # noqa: BLE001 — degrade to dense-only on any error
298        print(f"[startup] WARNING: could not open DocKG lexical store: {exc}")

380def _table_search(table, qvec, where: str, k: int) -> list[dict]:
381    """Run a cosine kNN search with a pre-filter; returns rows with ``_distance``.
382
383    Dispatches on the store type. The sqlite-vec backend is exact (recall 1.0)
384    and its ``where`` compiles to a true rowid-subquery prefilter. The LanceDB
385    path keeps the ``nprobes(128)`` stopgap that lifts IvfFlat recall@10 from
386    ~0.825 to ~0.99 (benchmarks/SQLITE_VEC_RESULTS.md).
387    """
388    from kg_utils.vector_backend import SqliteVecBackend
389
390    if isinstance(table, SqliteVecBackend):
391        return table.search(qvec, k, where=where)
392    return (
393        table.search(qvec)
394        .metric("cosine")
395        .where(where, prefilter=True)
396        .nprobes(128)
397        .limit(k)
398        .to_list()
399    )

402def _rrf_fuse(dense_ids: list[str], lex_ids: list[str], k: int) -> list[str]:
403    """Blend dense and lexical rank lists with reciprocal rank fusion (RRF).
404
405    Each channel contributes ``1 / (_RRF_K + rank)`` to a node's score, so a
406    chunk ranked highly by *either* cosine or BM25 floats up, while a chunk
407    ranked by both wins decisively.  This is what lets exact-term matches the
408    dense embedder buries (e.g. "Hell" in Dante's *Inferno*) seed the answer.
409
410    :param dense_ids: Node IDs ordered best-first by cosine distance.
411    :param lex_ids: Node IDs ordered best-first by BM25 lexical score.
412    :param k: Number of fused IDs to return.
413    :returns: Node IDs ordered best-first by fused RRF score.
414    """
415    scores: dict[str, float] = {}
416    for rank, nid in enumerate(dense_ids):
417        scores[nid] = scores.get(nid, 0.0) + 1.0 / (_RRF_K + rank)
418    for rank, nid in enumerate(lex_ids):
419        scores[nid] = scores.get(nid, 0.0) + 1.0 / (_RRF_K + rank)
420    return sorted(scores, key=lambda i: -scores[i])[:k]

423def _rows_to_hits(rows: list[dict], kg_name: str, kg_kind: str, min_score: float) -> list[dict]:
424    """Shape LanceDB rows into hit dicts (clean content hydrated separately).
425
426    The LanceDB ``text`` column holds the structured *embed-text* (``KIND:/TITLE:/
427    FILE:/TEXT:`` prefixed), not the clean passage — so ``content``/``summary`` are
428    left empty here and filled from SQLite by ``_attach_content`` / ``_attach_diary_fields``.
429    """
430    hits: list[dict] = []
431    for row in rows:
432        score = round(1.0 - float(row.get("_distance", 1.0)), 4)
433        if score < min_score:
434            continue
435        hits.append(
436            {
437                "kg_name": kg_name,
438                "kg_kind": kg_kind,
439                "node_id": row.get("id"),
440                "name": row.get("name") or row.get("title") or "",
441                "kind": row.get("kind", "chunk"),
442                "score": score,
443                "summary": "",
444                "source_path": row.get("file_path") or "",
445                "content": "",
446                "timestamp": None,
447            }
448        )
449    return hits

452def _attach_diary_fields(hits: list[dict]) -> None:
453    """Hydrate clean passage text and temporal ``timestamp`` from diary SQLite."""
454    by_kg: dict[str, list[dict]] = defaultdict(list)
455    for h in hits:
456        by_kg[h.get("kg_name", "")].append(h)
457    for slug, kg_hits in by_kg.items():
458        db_path = _KG_SQLITE.get(slug)
459        if not db_path or not Path(db_path).exists():
460            continue
461        ids = [h["node_id"] for h in kg_hits if h.get("node_id")]
462        if not ids:
463            continue
464        field_by_id: dict[str, tuple[str, str | None]] = {}
465        try:
466            with sqlite3.connect(str(db_path)) as con:
467                placeholders = ",".join("?" * len(ids))
468                rows = con.execute(
469                    f"SELECT id, text, timestamp FROM nodes WHERE id IN ({placeholders})", ids
470                )
471                for nid, text, ts in rows:
472                    field_by_id[nid] = (text or "", ts)
473        except Exception:  # noqa: BLE001
474            continue
475        for h in kg_hits:
476            text, ts = field_by_id.get(h.get("node_id", ""), ("", None))
477            h["content"] = text
478            h["summary"] = text
479            h["timestamp"] = ts

482def _semantic_search(
483    query: str,
484    k: int,
485    min_score: float = 0.0,
486    semantic_floor: float = 0.0,
487    genre_filter: str | None = None,
488) -> list[dict]:
489    """Hybrid (dense + lexical) search over the consolidated DocKG.
490
491    Ranks every chunk/section by its *own* relevance to the query — no graph-hop
492    expansion, so a query that names a book surfaces that book's passages on top
493    instead of letting them inherit a flat seed score from a few graph-expanded
494    neighbours.  Two channels run over the same in-scope subset and are blended
495    with reciprocal rank fusion:
496
497    * **dense** — cosine kNN over the LanceDB vector table;
498    * **lexical** — FTS5/BM25 over the SQLite ``nodes_fts`` index.
499
500    The lexical channel recovers exact-term hits that the embedder buries — e.g.
501    "circles of Hell" drifts semantically toward Dante's geometric *Paradiso*,
502    while BM25 pins the literal *Inferno* passages.  When no lexical index is
503    present (``_DOCKG_STORE is None``) the search degrades to pure dense ranking.
504    Content-kind and genre filters are pushed into *both* channels so the top-k
505    is computed over the eligible subset.
506
507    :param query: Natural-language query string.
508    :param k: Number of hits to return.
509    :param min_score: Drop hits whose cosine similarity is below this.
510    :param semantic_floor: If the best *dense* hit is below this, discard the set.
511    :param genre_filter: Restrict to a single genre subtree (e.g. ``sacred-texts``).
512    :returns: Hit dictionaries ranked best-first, shaped like ``hit_to_dict``.
513    """
514    if _DOCKG_TABLE is None:
515        return []
516    qvec = _embedder.embed_texts([query])[0]
517    where = "kind IN ('chunk', 'section') AND file_path NOT LIKE '%reference.md'"
518    if genre_filter:
519        # genre_filter is validated against _ALL_GENRES before this call.
520        where += f" AND file_path LIKE '{genre_filter}/%'"
521
522    # Dense channel — oversample so RRF has rank headroom for the lexical blend.
523    dense_rows = _table_search(_DOCKG_TABLE, qvec, where, k * 3)
524
525    # Lexical channel — BM25 over the same genre/kind-scoped subset (pushed down
526    # into the FTS5 SQL), returning chunk IDs ordered best-first.
527    lex_ids: list[str] = []
528    if _DOCKG_STORE is not None:
529        try:
530            lex_ids = _DOCKG_STORE.search_lexical(
531                query,
532                limit=k * 3,
533                file_prefixes=[f"{genre_filter}/"] if genre_filter else None,
534                node_kinds=("chunk", "section"),
535            )
536        except Exception as exc:  # noqa: BLE001 — degrade to dense-only on any error
537            print(f"[query] WARNING: lexical search failed, dense-only: {exc}")
538
539    row_by_id: dict[str, dict] = {r["id"]: r for r in dense_rows}
540    if lex_ids:
541        # Hydrate cosine rows for lexical-only IDs so every fused hit carries an
542        # honest cosine score and stays inside the genre/kind scope.
543        missing = [i for i in lex_ids if i and i not in row_by_id]
544        if missing:
545            id_list = ", ".join("'" + i.replace("'", "''") + "'" for i in missing)
546            lex_rows = _table_search(
547                _DOCKG_TABLE, qvec, f"{where} AND id IN ({id_list})", len(missing)
548            )
549            for r in lex_rows:
550                row_by_id.setdefault(r["id"], r)
551        lex_ids = [i for i in lex_ids if i in row_by_id]  # drop out-of-scope IDs
552        ordered_ids = _rrf_fuse([r["id"] for r in dense_rows], lex_ids, k)
553        ordered_rows = [row_by_id[i] for i in ordered_ids]
554    else:
555        ordered_rows = dense_rows[:k]
556
557    hits = _rows_to_hits(ordered_rows, "gutenberg", "KGKind.GUTENBERG", min_score)
558    _attach_content(hits)  # clean passage text from SQLite
559    for h in hits:
560        h["summary"] = h.get("content", "")
561    # The semantic floor gauges whether the KG is relevant at all, so test it
562    # against the best *dense* cosine (the lexical channel can surface a literal
563    # match with modest cosine that should not, by itself, keep a stale set).
564    if semantic_floor > 0.0:
565        best_dense = round(1.0 - float(dense_rows[0]["_distance"]), 4) if dense_rows else 0.0
566        if best_dense < semantic_floor:
567            return []
568    return hits

571def _semantic_search_diaries(
572    query: str,
573    k: int,
574    min_score: float = 0.0,
575    semantic_floor: float = 0.0,
576) -> list[dict]:
577    """Pure cosine search across the DiaryKG vector tables, merged by score.
578
579    Gives diaries the same true-cosine scoring as the books, so a unified
580    (``all``) query ranks both corpora on one comparable scale instead of letting
581    the diaries' graph-expanded plateau scores dominate.
582
583    :param query: Natural-language query string.
584    :param k: Number of hits to return across all diaries.
585    :param min_score: Drop hits whose cosine similarity is below this.
586    :param semantic_floor: If the best hit is below this, discard the whole set.
587    :returns: Hit dictionaries ranked best-first across all diary KGs.
588    """
589    if not _DIARY_TABLES:
590        return []
591    qvec = _embedder.embed_texts([query])[0]
592    where = "kind IN ('chunk', 'section')"
593    hits: list[dict] = []
594    for slug, table in _DIARY_TABLES.items():
595        rows = _table_search(table, qvec, where, k)
596        hits.extend(_rows_to_hits(rows, slug, "KGKind.DIARY", min_score))
597    hits.sort(key=lambda h: h["score"], reverse=True)
598    hits = hits[:k]
599    _attach_diary_fields(hits)  # clean text + timestamp from each diary's SQLite
600    if semantic_floor > 0.0 and hits and hits[0]["score"] < semantic_floor:
601        return []
602    return hits

605def _dockg_store_ro():
606    """Return a ``GraphStore`` for read-only node lookups (chapters/sections).
607
608    Reuses ``_DOCKG_STORE`` if the FTS-gated lexical channel already opened one;
609    otherwise lazily opens and caches a plain store, since chapter/section reads
610    don't require an FTS5 index.
611
612    :returns: An open ``GraphStore``, or ``None`` if the DocKG SQLite is missing.
613    """
614    global _DOCKG_STORE_RO
615    if _DOCKG_STORE is not None:
616        return _DOCKG_STORE
617    if _DOCKG_STORE_RO is None and _DOCKG_SQLITE.exists():
618        from doc_kg.store import GraphStore
619
620        _DOCKG_STORE_RO = GraphStore(_DOCKG_SQLITE)
621    return _DOCKG_STORE_RO

624def _resolve_book_file_path(genre: str, book: str) -> str | None:
625    """Look up the content ``.md`` file path for a ``<genre>/<book>`` pair.
626
627    Each book directory has exactly one content document plus a ``reference.md``;
628    this is a single prefix lookup that ``GraphStore.query_nodes`` doesn't support
629    directly (it only matches ``file_path`` exactly), so it stays raw SQL.
630
631    :param genre: Genre slug, e.g. ``"american-literature"``.
632    :param book: Book directory name, e.g. ``"Adventures of Huckleberry Finn"``.
633    :returns: The node ``file_path`` for the book's content document, or ``None``.
634    """
635    store = _dockg_store_ro()
636    if store is None:
637        return None
638    row = store.con.execute(
639        "SELECT file_path FROM nodes WHERE kind='document' AND file_path LIKE ? "
640        "AND file_path NOT LIKE '%reference.md' LIMIT 1",
641        (f"{genre}/{book}/%",),
642    ).fetchone()
643    return row[0] if row else None

658def _corpus_stats() -> dict:
659    """Return live corpus totals for the chat UI header (no hardcoded counts).
660
661    Books and genres come from ``catalog.json``; node/edge totals from the
662    consolidated ``graph.sqlite``; diary count from the registered DiaryKG
663    tables. Returns zeros for any store that can't be opened.
664
665    :returns: ``{"books", "genres", "diaries", "nodes", "edges", "embed_model"}``.
666    """
667    genres = {meta.get("genre", "") for meta in _catalog.values() if meta.get("genre")}
668    nodes = edges = 0
669    if _DOCKG_SQLITE.exists():
670        try:
671            with sqlite3.connect(_DOCKG_SQLITE) as con:
672                nodes = con.execute("SELECT COUNT(*) FROM nodes").fetchone()[0]
673                edges = con.execute("SELECT COUNT(*) FROM edges").fetchone()[0]
674        except sqlite3.Error:
675            pass
676    return {
677        "books": len(_catalog),
678        "genres": len(genres),
679        "diaries": len(_DIARY_TABLES),
680        "nodes": nodes,
681        "edges": edges,
682        "embed_model": EMBED_MODEL,
683    }

706def _get_chapters(genre: str, book: str) -> dict:
707    """List a book's chapters (from ``section`` nodes, or ``chunk.chapter`` as fallback).
708
709    :param genre: Genre slug.
710    :param book: Book directory name.
711    :returns: ``{"book", "chapters": [{"id", "title", "index"}, ...]}``, or an
712        ``"error"`` key if the book/store can't be resolved.
713    """
714    store = _dockg_store_ro()
715    if store is None:
716        return {"error": "DocKG store unavailable"}
717    file_path = _resolve_book_file_path(genre, book)
718    if file_path is None:
719        return {"error": f"book not found: {genre}/{book}"}
720
721    sections = store.query_nodes(kinds=["section"], file_path=file_path)
722    if sections:
723        chapters = [
724            {"id": s["id"], "title": s.get("title") or s.get("name"), "index": i}
725            for i, s in enumerate(sections)
726        ]
727        return {"book": book, "chapters": chapters}
728
729    # Verse-chunked genres (sacred-texts) may carry no section nodes — fall back
730    # to grouping chunks by their `chapter` column.
731    chunks = store.query_nodes(kinds=["chunk"], file_path=file_path)
732    seen_chapters: list[int] = []
733    for c in chunks:
734        ch = c.get("chapter")
735        if ch is not None and ch not in seen_chapters:
736            seen_chapters.append(ch)
737    chapters = [
738        {"id": f"chapter:{ch}", "title": f"Chapter {ch}", "index": i}
739        for i, ch in enumerate(seen_chapters)
740    ]
741    return {"book": book, "chapters": chapters}

744def _get_chapter(genre: str, book: str, section_id: str) -> dict:
745    """Reconstruct one chapter's text by concatenating its chunks in order.
746
747    :param genre: Genre slug.
748    :param book: Book directory name.
749    :param section_id: A chapter id from :func:`_get_chapters` — either a
750        ``section`` node id, or a synthetic ``"chapter:<n>"`` id from the
751        chunk-grouping fallback.
752    :returns: ``{"title", "text", "index", "total", "prev_id", "next_id"}``, or
753        an ``"error"`` key if the book/section can't be resolved.
754    """
755    store = _dockg_store_ro()
756    if store is None:
757        return {"error": "DocKG store unavailable"}
758    file_path = _resolve_book_file_path(genre, book)
759    if file_path is None:
760        return {"error": f"book not found: {genre}/{book}"}
761
762    sections = store.query_nodes(kinds=["section"], file_path=file_path)
763    if sections:
764        index = next((i for i, s in enumerate(sections) if s["id"] == section_id), None)
765        if index is None:
766            return {"error": f"unknown section: {section_id}"}
767        start = sections[index]["char_start"]
768        end = sections[index + 1]["char_start"] if index + 1 < len(sections) else None
769        chunks = store.query_nodes(kinds=["chunk"], file_path=file_path)
770        text = "\n\n".join(
771            c["text"]
772            for c in chunks
773            if c["char_start"] >= start and (end is None or c["char_start"] < end)
774        )
775        return {
776            "title": sections[index].get("title") or sections[index].get("name"),
777            "text": text,
778            "index": index,
779            "total": len(sections),
780            "prev_id": sections[index - 1]["id"] if index > 0 else None,
781            "next_id": sections[index + 1]["id"] if index + 1 < len(sections) else None,
782        }
783
784    # Chunk-grouping fallback (verse-chunked genres).
785    if not section_id.startswith("chapter:"):
786        return {"error": f"unknown section: {section_id}"}
787    chapter_num = int(section_id.split(":", 1)[1])
788    chunks = store.query_nodes(kinds=["chunk"], file_path=file_path)
789    seen_chapters: list[int] = []
790    for c in chunks:
791        ch = c.get("chapter")
792        if ch is not None and ch not in seen_chapters:
793            seen_chapters.append(ch)
794    if chapter_num not in seen_chapters:
795        return {"error": f"unknown chapter: {chapter_num}"}
796    index = seen_chapters.index(chapter_num)
797    text = "\n\n".join(c["text"] for c in chunks if c.get("chapter") == chapter_num)
798    prev_ch = seen_chapters[index - 1] if index > 0 else None
799    next_ch = seen_chapters[index + 1] if index + 1 < len(seen_chapters) else None
800    return {
801        "title": f"Chapter {chapter_num}",
802        "text": text,
803        "index": index,
804        "total": len(seen_chapters),
805        "prev_id": f"chapter:{prev_ch}" if prev_ch is not None else None,
806        "next_id": f"chapter:{next_ch}" if next_ch is not None else None,
807    }

810def _enrich_catalog(hits: list[dict]) -> None:
811    """Join author/title/genre from catalog.json onto DocKG hits."""
812    for h in hits:
813        if h.get("kg_kind") in ("KGKind.GUTENBERG", "gutenberg"):
814            src = h.get("source_path", "")
815            parts = src.split("/")
816            if len(parts) >= 2:
817                key = f"{parts[0]}/{parts[1]}"
818                meta = _catalog.get(key, {})
819                h["genre"] = meta.get("genre") or parts[0]
820                h["title"] = meta.get("title") or parts[1]
821                h["author"] = meta.get("author")
822        # Diary hits: fall back to static map keyed by slug.
823        diary = _DIARY_META.get(h.get("kg_name", ""))
824        if diary and not h.get("author"):
825            h["genre"] = diary["genre"]
826            h["title"] = diary["title"]
827            h["author"] = diary["author"]

835def handler(job: dict) -> dict:
836    """RunPod serverless entry point: search the corpus and optionally synthesise an answer.
837
838    :param job: RunPod job dict; ``job["input"]`` holds the request schema described
839        in this module's docstring (query, corpus, k, min_score, semantic_floor,
840        synthesize, model, secret, op).
841    :returns: A result dict with hits, timing, and optional synthesis text, or
842        ``{"error": ...}`` on validation/auth failure.
843    """
844    inp = job.get("input", {})
845
846    if HANDLER_SECRET and inp.get("secret") != HANDLER_SECRET:
847        return {"error": "unauthorized"}
848
849    aux_result = handle_aux_ops(inp, _synth_for_backend, _image_for_backend)
850    if aux_result is not None:
851        return aux_result
852
853    op = inp.get("op", "")
854    if op == "stats":
855        return _corpus_stats()
856    if op == "list_genres":
857        return _list_genres()
858    if op == "list_books":
859        return _list_books(inp.get("genre", ""))
860    if op == "get_chapters":
861        return _get_chapters(inp.get("genre", ""), inp.get("book", ""))
862    if op == "get_chapter":
863        return _get_chapter(inp.get("genre", ""), inp.get("book", ""), inp.get("section_id", ""))
864
865    query = inp.get("query", "").strip()
866    corpus = inp.get("corpus", "all")
867    k = max(1, int(inp.get("k", 8)))
868    min_score = float(inp.get("min_score", 0.0))
869    semantic_floor = float(inp.get("semantic_floor", 0.0))
870    synthesize = bool(inp.get("synthesize", False))
871    model = (inp.get("model") or "").strip() or None
872
873    if not query:
874        return {"error": "query is required"}
875
876    genre_filter: str | None = None
877    if corpus in _ALL_GENRES:
878        genre_filter = corpus
879    elif corpus not in ("all", "diary", "gutenberg"):
880        return {
881            "error": (
882                f"unknown corpus {corpus!r}; choose: all, gutenberg, diary, "
883                f"or a genre name ({', '.join(sorted(_ALL_GENRES))})"
884            )
885        }
886
887    t0_search = time.perf_counter()
888
889    if corpus == "diary":
890        # Diaries only — semantic-first across the DiaryKG vector tables.
891        hits = _semantic_search_diaries(
892            query, k=k, min_score=min_score, semantic_floor=semantic_floor
893        )
894        _enrich_catalog(hits)
895        kgs_queried = len(_DIARY_TABLES)
896    else:
897        # Semantic-first: rank chunks by their own cosine distance (no graph-hop
898        # expansion), so a query that names a book surfaces that book on top.
899        hits = _semantic_search(
900            query,
901            k=k,
902            min_score=min_score,
903            semantic_floor=semantic_floor,
904            genre_filter=genre_filter,
905        )
906        _enrich_catalog(hits)
907        kgs_queried = 1  # the consolidated 'gutenberg' DocKG
908        if corpus == "all":
909            # Fold in the diaries on the *same* true-cosine scale, then re-rank
910            # the merged set by score so neither corpus's scoring dominates.
911            dhits = _semantic_search_diaries(
912                query, k=k, min_score=min_score, semantic_floor=semantic_floor
913            )
914            _enrich_catalog(dhits)
915            hits = sorted(hits + dhits, key=lambda h: h.get("score", 0.0), reverse=True)[:k]
916            kgs_queried += len(_DIARY_TABLES)
917
918    search_ms = (time.perf_counter() - t0_search) * 1000
919    print(f"[query] {len(hits)} matching results found in {search_ms:.0f}ms")
920
921    synthesis = None
922    synthesis_ms: float | None = None
923    active_synth = _synth_for_backend(inp.get("backend", ""))
924    if synthesize:
925        t0_synth = time.perf_counter()
926        synthesis = active_synth.synthesize_rag(query, hits, model=model, max_k=SYNTH_MAX_K)
927        synthesis_ms = (time.perf_counter() - t0_synth) * 1000
928        print(f"[query] synthesis returned in {synthesis_ms:.0f}ms")
929
930    return {
931        "query": query,
932        "corpus": corpus,
933        "total_hits": len(hits),
934        "kgs_queried": kgs_queried,
935        "hits": hits,
936        "search_ms": round(search_ms),
937        "synthesis": synthesis,
938        "synthesis_ms": round(synthesis_ms) if synthesis_ms is not None else None,
939        "model": (model or active_synth._cfg.resolved_model()) if synthesize else None,
940    }
